-- Step 1: Filter Q3 enterprise cohorts with >25% discount threshold
WITH policy_breaches AS (
    SELECT 
        order_id,
        customer_segment,
        sku_name,
        gross_revenue,
        applied_discount_pct,
        (applied_discount_pct - 0.25) * gross_revenue AS margin_drag,
        CASE WHEN applied_discount_pct > 0.25 THEN TRUE ELSE FALSE END AS policy_violation_flag
    FROM enterprise_saas_orders
    WHERE fiscal_quarter = 'Q3-2024'
      AND applied_discount_pct > 0.25
)
SELECT 
    order_id,
    customer_segment,
    sku_name,
    gross_revenue,
    applied_discount_pct,
    policy_violation_flag
FROM policy_breaches
ORDER BY applied_discount_pct DESC;
